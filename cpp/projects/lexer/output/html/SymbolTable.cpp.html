   24 int SymbolTable::insertOrGet(std::vector<std::string> &values,
   25                              std::unordered_map<std::string, int> &indexes,
   26                              const std::string &value)
   27 {
   28     auto iterator = indexes.find(value);
   29 
   30     if (iterator != indexes.end())
   31         return iterator->second;
   32 
   33     int index = static_cast<int>(values.size());
   34 
   35     values.push_back(value);
   36     indexes.emplace(value, index);
   37 
   38     return index;
   39 }

   41 void SymbolTable::writeToFile(const std::string &path) const
   42 {
   43     fs::path outputPath(path);
   44 
   45     if (!outputPath.parent_path().empty())
   46         fs::create_directories(outputPath.parent_path());
   47 
   48     std::ofstream output(path);
   49 
   50     if (!output.is_open())
   51         throw std::runtime_error("cannot create symbol table file: " + path);
   52 
   53     output << "Identifier Table:\n";
   54     for (std::size_t i = 0; i < identifiers_.size(); i++)
   55         output << i << ": " << identifiers_[i] << '\n';
   56 
   57     output << "\nConstant Table:\n";
   58     for (std::size_t i = 0; i < constants_.size(); i++)
   59         output << i << ": " << constants_[i] << '\n';
   60 
   61     output << "\nLabel Table:\n";
   62     for (std::size_t i = 0; i < labels_.size(); i++)
   63         output << i << ": " << labels_[i] << '\n';
   64 }
