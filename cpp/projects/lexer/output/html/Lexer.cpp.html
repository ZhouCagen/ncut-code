   15 bool isIdentifierStart(char ch)
   16 {
   17     return std::isalpha(static_cast<unsigned char>(ch));
   18 }
   19 
   20 bool isIdentifierBody(char ch)
   21 {
   22     return std::isalnum(static_cast<unsigned char>(ch)) || ch == '_';
   23 }
   24 
   25 bool isDigit(char ch)
   26 {
   27     return std::isdigit(static_cast<unsigned char>(ch));
   28 }
   29 
   30 bool isKeyword(const std::string &word)
   31 {
   32     static const std::unordered_set<std::string> keywords = {"IF", "THEN", "ELSE", "GOTO"};
   33 
   34     return keywords.contains(word);
   35 }
   36 
   37 bool isDoubleOperator(const std::string &value)
   38 {
   39     static const std::unordered_set<std::string> operators = {">=", "<=", "<>"};
   40 
   41     return operators.contains(value);
   42 }
   43 
   44 bool isSingleOperator(char ch)
   45 {
   46     static const std::set<char> operators = {'+', '-', '*', '/', '=', '>', '<', '^'};
   47 
   48     return operators.contains(ch);
   49 }
   50 
   51 bool isPunctuation(char ch)
   52 {
   53     static const std::set<char> punctuations = {'(', ')', ':', ';'};
   54 
   55     return punctuations.contains(ch);
   56 }

   63 std::vector<Token> Lexer::scan(std::string_view source)
   64 {
   65     std::vector<Token> tokens;
   66 
   67     std::size_t i = 0;
   68     std::size_t line = 1;
   69     std::size_t column = 1;
   70 
   71     bool previousTokenIsGoto = false;
   72 
   73     auto currentLocation = [&]() { return SourceLocation{.line = line, .column = column}; };
   74 
   75     auto advance = [&]()
   76     {
   77         char ch = source[i];
   78         i++;
   79 
   80         if (ch == '\n')
   81         {
   82             line++;
   83             column = 1;
   84         }
   85         else
   86         {
   87             column++;
   88         }
   89 
   90         return ch;
   91     };
   92 
   93     while (i < source.size())
   94     {
   95         char ch = source[i];
   96 
   97         if (std::isspace(static_cast<unsigned char>(ch)))
   98         {
   99             advance();
  100             continue;
  101         }
  102 
  103         SourceLocation startLocation = currentLocation();
  104 
  105         if (ch == '#')
  106         {
  107             advance();
  108 
  109             tokens.push_back(Token{.type = TokenType::End,
  110                                    .lexeme = "EOF",
  111                                    .location = startLocation,
  112                                    .tableIndex = std::nullopt});
  113 
  114             break;
  115         }
  116 
  117         State state;
  118 
  119         if (isIdentifierStart(ch))
  120             state = State::Identifier;
  121         else if (isDigit(ch))
  122             state = State::Number;
  123         else
  124             state = State::OperatorOrPunctuation;
  125 
  126         switch (state)
  127         {
  128             case State::Identifier:
  129             {
  130                 std::string word;
  131 
  132                 while (i < source.size() && isIdentifierStart(source[i]))
  133                     word += advance();
  134 
  135                 if (i < source.size() && isKeyword(word))
  136                 {
  137                     tokens.push_back(Token{.type = TokenType::Keyword,
  138                                            .lexeme = word,
  139                                            .location = startLocation,
  140                                            .tableIndex = std::nullopt});
  141                 }
  142                 else
  143                 {
  144                     int tableIndex = symbolTable_.insertIdentifier(word);
  145                     tokens.push_back(Token{.type = TokenType::Identifier,
  146                                            .lexeme = word,
  147                                            .location = startLocation,
  148                                            .tableIndex = tableIndex});
  149                 }
  150                 break;
  151             }
  152 
  153             case State::Number:
  154             {
  155                 std::string number;
  156 
  157                 while (i < source.size() && isDigit(source[i]))
  158                     number += advance();
  159 
  160                 if (i < source.size() && isIdentifierStart(source[i]))
  161                 {
  162                     throw makeError(startLocation,
  163                                     "number cannot be followed directly by identifier character: " +
  164                                         number);
  165                 }
  166 
  167                 bool isLabel = false;
  168 
  169                 if (previousTokenIsGoto)
  170                     isLabel = true;
  171 
  172                 if (i < source.size() && source[i] == ':')
  173                     isLabel = true;
  174 
  175                 if (isLabel)
  176                 {
  177                     int tableIndex = symbolTable_.insertLabel(number);
  178 
  179                     tokens.push_back(Token{.type = TokenType::Label,
  180                                            .lexeme = number,
  181                                            .location = startLocation,
  182                                            .tableIndex = static_cast<std::size_t>(tableIndex)});
  183                 }
  184                 else
  185                 {
  186                     int tableIndex = symbolTable_.insertConstant(number);
  187 
  188                     tokens.push_back(Token{.type = TokenType::Constant,
  189                                            .lexeme = number,
  190                                            .location = startLocation,
  191                                            .tableIndex = static_cast<std::size_t>(tableIndex)});
  192                 }
  193 
  194                 previousTokenIsGoto = false;
  195                 break;
  196             }
  197 
  198             case State::OperatorOrPunctuation:
  199             {
  200                 if (i + 1 < source.size())
  201                 {
  202                     std::string twoChars;
  203                     twoChars += source[i];
  204                     twoChars += source[i + 1];
  205 
  206                     if (isDoubleOperator(twoChars))
  207                     {
  208                         advance();
  209                         advance();
  210 
  211                         tokens.push_back(Token{.type = TokenType::Operator,
  212                                                .lexeme = twoChars,
  213                                                .location = startLocation,
  214                                                .tableIndex = std::nullopt});
  215 
  216                         previousTokenIsGoto = false;
  217                         break;
  218                     }
  219                 }
  220 
  221                 if (isSingleOperator(ch))
  222                 {
  223                     std::string value;
  224                     value += advance();
  225 
  226                     tokens.push_back(Token{.type = TokenType::Operator,
  227                                            .lexeme = value,
  228                                            .location = startLocation,
  229                                            .tableIndex = std::nullopt});
  230 
  231                     previousTokenIsGoto = false;
  232                     break;
  233                 }
  234 
  235                 if (isPunctuation(ch))
  236                 {
  237                     std::string value;
  238                     value += advance();
  239 
  240                     tokens.push_back(Token{.type = TokenType::Punctuation,
  241                                            .lexeme = value,
  242                                            .location = startLocation,
  243                                            .tableIndex = std::nullopt});
  244 
  245                     previousTokenIsGoto = false;
  246                     break;
  247                 }
  248 
  249                 throw makeError(startLocation, std::string("unknown character: ") + ch);
  250             }
  251         }
  252     }
  253 
  254     return tokens;
  255 }

  257 void Lexer::writeTokens(const std::vector<Token> &tokens, const std::string &path) const
  258 {
  259     fs::path outputPath(path);
  260 
  261     if (!outputPath.parent_path().empty())
  262         fs::create_directories(outputPath.parent_path());
  263 
  264     std::ofstream output(path);
  265 
  266     if (!output.is_open())
  267         throw std::runtime_error("cannot create token file: " + path);
  268 
  269     for (const auto &token : tokens)
  270     {
  271         if (token.type == TokenType::End)
  272         {
  273             output << "(#,EOF)\n";
  274         }
  275         else
  276         {
  277             output << '(' << tokenTypeCode(token.type) << ',' << token.lexeme << ")\n";
  278         }
  279     }
  280 }

  282 std::runtime_error Lexer::makeError(const SourceLocation &location,
  283                                     const std::string &message) const
  284 {
  285     return std::runtime_error("lexical error at line " + std::to_string(location.line) +
  286                               ", column " + std::to_string(location.column) + ": " + message);
