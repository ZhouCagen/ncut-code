   10 std::string Preprocessor::preprocessFile(const std::string &inputPath,
   11                                          const std::string &outputPath) const
   12 {
   13     std::string joined = readAndJoinContinuedLines(inputPath);
   14     std::string noComment = removeComments(joined);
   15     std::string normalized = normalizeSpaces(noComment);
   16 
   17     if (normalized.empty() || normalized.back() != '#')
   18         normalized += '#';
   19 
   20     writeFile(outputPath, normalized);
   21 
   22     return normalized;
   23 }

   25 std::string Preprocessor::readAndJoinContinuedLines(const std::string &inputPath) const
   26 {
   27     std::ifstream input(inputPath);
   28     if (!input.is_open())
   29         throw std::runtime_error("cannot open source file: " + inputPath);
   30 
   31     std::string result;
   32     std::string line;
   33     int continuationCount = 0;
   34 
   35     while (std::getline(input, line))
   36     {
   37         if (!line.empty() && line.back() == '\r')
   38             line.pop_back();
   39 
   40         if (!line.empty() && line.back() == '\\')
   41         {
   42             line.pop_back();
   43             result += line;
   44             continuationCount++;
   45 
   46             if (continuationCount > 3)
   47                 throw std::runtime_error("continuation lines exceed 3");
   48         }
   49         else
   50         {
   51             result += line;
   52             result += '\n';
   53             continuationCount = 0;
   54         }
   55     }
   56     return result;
   57 }

   59 std::string Preprocessor::removeComments(const std::string &source) const
   60 {
   61     std::string result;
   62     bool inBlockComment = false;
   63 
   64     for (std::size_t i = 0; i < source.size(); i++)
   65     {
   66         if (!inBlockComment && i + 1 < source.size() && source[i] == '/' && source[i + 1] == '*')
   67         {
   68             inBlockComment = true;
   69             result += ' ';
   70             i++;
   71             continue;
   72         }
   73 
   74         if (inBlockComment && i + 1 < source.size() && source[i] == '*' && source[i + 1] == '/')
   75         {
   76             inBlockComment = false;
   77             i++;
   78             continue;
   79         }
   80 
   81         if (!inBlockComment)
   82             result += source[i];
   83     }
   84 
   85     if (inBlockComment)
   86         throw std::runtime_error("unclosed block comment");
   87 
   88     return result;
   89 }

   91 std::string Preprocessor::normalizeSpaces(const std::string &source) const
   92 {
   93     std::string result;
   94     bool previousIsSpace = false;
   95 
   96     for (char ch : source)
   97     {
   98         if (std::isspace(static_cast<unsigned char>(ch)))
   99         {
  100             if (!previousIsSpace && !result.empty())
  101             {
  102                 result += ' ';
  103                 previousIsSpace = true;
  104             }
  105         }
  106         else
  107         {
  108             result += ch;
  109             previousIsSpace = false;
  110         }
  111     }
  112 
  113     while (!result.empty() && result.back() == ' ')
  114         result.pop_back();
  115 
  116     return result;
  117 }

  119 void Preprocessor::writeFile(const std::string &outputPath, const std::string &content) const
  120 {
  121     fs::path path(outputPath);
  122 
  123     if (!path.parent_path().empty())
  124         fs::create_directories(path.parent_path());
  125 
  126     std::ofstream output(outputPath);
  127 
  128     if (!output.is_open())
  129         throw std::runtime_error("cannot create output file: " + outputPath);
  130 
  131     output << content;
  132 }
