1 #pragma once
    2 
    3 #include <cstddef>
    4 #include <optional>
    5 #include <string>
    6 
    7 enum class TokenType
    8 {
    9     Keyword,
   10     Identifier,
   11     Constant,
   12     Label,
   13     Operator,
   14     Punctuation,
   15     End
   16 };
   17 
   18 struct SourceLocation
   19 {
   20     std::size_t line = 1;
   21     std::size_t column = 1;
   22 };
   23 
   24 struct Token
   25 {
   26     TokenType type;
   27     std::string lexeme;
   28     SourceLocation location;
   29     std::optional<std::size_t> tableIndex;
   30 };
   31 
   32 inline char tokenTypeCode(TokenType type)
   33 {
   34     switch (type)
   35     {
   36         case TokenType::Keyword:
   37             return 'K';
   38         case TokenType::Identifier:
   39             return 'I';
   40         case TokenType::Constant:
   41             return 'C';
   42         case TokenType::Label:
   43             return 'L';
   44         case TokenType::Operator:
   45             return 'O';
   46         case TokenType::Punctuation:
   47             return 'P';
   48         case TokenType::End:
   49             return '#';
   50     }
   51 
   52     return '?';
   53 }
   54 
   55 inline std::string_view tokenTypeName(TokenType type)
   56 {
   57     switch (type)
   58     {
   59         case TokenType::Keyword:
   60             return "Keyword";
   61         case TokenType::Identifier:
   62             return "Identifier";
   63         case TokenType::Constant:
   64             return "Constant";
   65         case TokenType::Label:
   66             return "Label";
   67         case TokenType::Operator:
   68             return "Operator";
   69         case TokenType::Punctuation:
   70             return "Punctuation";
   71         case TokenType::End:
   72             return "End";
   73     }
   74 
   75     return "Unknown";
   76 }
