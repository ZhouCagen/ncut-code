1 #include "Lexer.hpp"
    2 #include "Preprocessor.hpp"
    3 #include "SymbolTable.hpp"
    4 #include "Token.hpp"
    5 
    6 #include <exception>
    7 #include <filesystem>
    8 #include <iostream>
    9 #include <string>
   10 #include <vector>
   11 
   12 namespace fs = std::filesystem;
   13 
   14 int main(int argc, char *argv[])
   15 {
   16     try
   17     {
   18         std::string sourcePath = "source.txt";
   19 
   20         if (argc >= 2)
   21             sourcePath = argv[1]; // ./lexer test.txt
   22 
   23         const std::string outputDirectory = "output";
   24         const std::string preprocessedPath = outputDirectory + "/preprocessed.txt";
   25         const std::string tokenPath = outputDirectory + "/tokens.txt";
   26         const std::string tablePath = outputDirectory + "/tables.txt";
   27 
   28         fs::create_directories(outputDirectory);
   29 
   30         Preprocessor preprocessor;
   31         std::string preprocessed = preprocessor.preprocessFile(sourcePath, preprocessedPath);
   32 
   33         SymbolTable symbolTable;
   34         Lexer lexer(symbolTable);
   35 
   36         std::vector<Token> tokens = lexer.scan(preprocessed);
   37 
   38         lexer.writeTokens(tokens, tokenPath);
   39         symbolTable.writeToFile(tablePath);
   40 
   41         std::cout << "词法分析完成\n";
   42         std::cout << "源程序文件: " << sourcePath << '\n';
   43         std::cout << "预处理文件: " << preprocessedPath << '\n';
   44         std::cout << "二元式文件: " << tokenPath << '\n';
   45         std::cout << "符号表文件: " << tablePath << '\n';
   46     }
   47     catch (const std::exception &error)
   48     {
   49         std::cerr << "error: " << error.what() << '\n';
   50         return 1;
   51     }
   52 
   53     return 0;
   54 }
