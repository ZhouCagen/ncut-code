   13 namespace fs = std::filesystem;
   14 
   15 std::string readFile(const std::string &path)
   16 {
   17     std::ifstream input(path);
   18 
   19     if (!input.is_open())
   20         throw std::runtime_error("cannot open input file: " + path);
   21 
   22     std::ostringstream buffer;
   23     buffer << input.rdbuf();
   24 
   25     return buffer.str();
   26 }
   27 
   28 void writeFailureResult(const std::string &path, const std::string &message)
   29 {
   30     fs::path outputPath(path);
   31 
   32     if (!outputPath.parent_path().empty())
   33         fs::create_directories(outputPath.parent_path());
   34 
   35     std::ofstream output(path);
   36 
   37     if (!output.is_open())
   38         throw std::runtime_error("cannot create parser result file: " + path);
   39 
   40     output << "Syntax analysis failed.\n";
   41     output << "Error: " << message << '\n';
   42 }
   43 
   44 void writeScannedTokens(const std::vector<Token> &tokens, const std::string &path)
   45 {
   46     fs::path outputPath(path);
   47 
   48     if (!outputPath.parent_path().empty())
   49         fs::create_directories(outputPath.parent_path());
   50 
   51     std::ofstream output(path);
   52 
   53     if (!output.is_open())
   54         throw std::runtime_error("cannot create token debug file: " + path);
   55 
   56     output << "Scanned tokens:\n";
   57 
   58     for (const auto &token : tokens)
   59     {
   60         output << token.location.line << ':' << token.location.column << "  "
   61                << tokenTypeName(token.type) << "  " << token.lexeme << '\n';
   62     }
   63 }
   64 
   65 int main(int argc, char *argv[])
   66 {
   67     const std::string outputDirectory = "output";
   68     const std::string reductionPath = outputDirectory + "/reductions.txt";
   69     const std::string resultPath = outputDirectory + "/parser_result.txt";
   70     const std::string tokenDebugPath = outputDirectory + "/scanned_tokens.txt";
   71 
   72     try
   73     {
   74         std::string inputPath = "preprocessed.txt";
   75 
   76         if (argc >= 2)
   77             inputPath = argv[1];
   78 
   79         fs::create_directories(outputDirectory);
   80 
   81         std::string source = readFile(inputPath);
   82 
   83         Scanner scanner(source);
   84         std::vector<Token> tokens = scanner.scan();
   85 
   86         writeScannedTokens(tokens, tokenDebugPath);
   87 
   88         Parser parser(tokens);
   89         parser.parseProgram();
   90 
   91         parser.writeReductions(reductionPath);
   92         parser.writeResult(resultPath);
   93 
   94         std::cout << "语法分析完成\n";
   95         std::cout << "输入文件: " << inputPath << '\n';
   96         std::cout << "Token调试文件: " << tokenDebugPath << '\n';
   97         std::cout << "归约序列文件: " << reductionPath << '\n';
   98         std::cout << "分析结果文件: " << resultPath << '\n';
   99     }
  100     catch (const std::exception &error)
  101     {
  102         try
  103         {
  104             writeFailureResult(resultPath, error.what());
  105         }
  106         catch (...)
  107         {
  108         }
  109 
  110         std::cerr << "error: " << error.what() << '\n';
  111         return 1;
  112     }
  113 
  114     return 0;
  115 }
