1 #include "Scanner.hpp"
    2 
    3 #include <cctype>
    4 #include <stdexcept>
    5 #include <string>
    6 
    7 namespace
    8 {
    9 enum class State
   10 {
   11     Identifier,
   12     Number,
   13     OperatorOrPunctuation
   14 };
   15 } // namespace
   16 
   17 Scanner::Scanner(std::string_view source) : source_(source)
   18 {
   19 }
   20 
   21 std::vector<Token> Scanner::scan()
   22 {
   23     std::vector<Token> tokens;
   24 
   25     std::size_t i = 0;
   26     std::size_t line = 1;
   27     std::size_t column = 1;
   28 
   29     bool previousTokenIsGoto = false;
   30 
   31     auto advance = [&]()
   32     {
   33         char ch = source_[i];
   34         i++;
   35 
   36         if (ch == '\n')
   37         {
   38             line++;
   39             column = 1;
   40         }
   41         else
   42         {
   43             column++;
   44         }
   45 
   46         return ch;
   47     };
   48 
   49     while (i < source_.size())
   50     {
   51         char ch = source_[i];
   52 
   53         if (std::isspace(static_cast<unsigned char>(ch)))
   54         {
   55             advance();
   56             continue;
   57         }
   58 
   59         SourceLocation location{.line = line, .column = column};
   60 
   61         if (ch == '#')
   62         {
   63             advance();
   64 
   65             tokens.push_back(Token{.type = TokenType::End, .lexeme = "#", .location = location});
   66             break;
   67         }
   68 
   69         State state;
   70 
   71         if (std::isalpha(static_cast<unsigned char>(ch)))
   72             state = State::Identifier;
   73         else if (std::isdigit(static_cast<unsigned char>(ch)))
   74             state = State::Number;
   75         else
   76             state = State::OperatorOrPunctuation;
   77 
   78         switch (state)
   79         {
   80             case State::Identifier:
   81             {
   82                 std::string word;
   83 
   84                 while (i < source_.size() &&
   85                        (std::isalnum(static_cast<unsigned char>(source_[i])) || source_[i] == '_'))
   86                 {
   87                     word += advance();
   88                 }
   89 
   90                 if (word == "IF")
   91                 {
   92                     tokens.push_back(
   93                         Token{.type = TokenType::If, .lexeme = word, .location = location});
   94                     previousTokenIsGoto = false;
   95                 }
   96                 else if (word == "THEN")
   97                 {
   98                     tokens.push_back(
   99                         Token{.type = TokenType::Then, .lexeme = word, .location = location});
  100                     previousTokenIsGoto = false;
  101                 }
  102                 else if (word == "ELSE")
  103                 {
  104                     tokens.push_back(
  105                         Token{.type = TokenType::Else, .lexeme = word, .location = location});
  106                     previousTokenIsGoto = false;
  107                 }
  108                 else if (word == "GOTO")
  109                 {
  110                     tokens.push_back(
  111                         Token{.type = TokenType::Goto, .lexeme = word, .location = location});
  112                     previousTokenIsGoto = true;
  113                 }
  114                 else
  115                 {
  116                     tokens.push_back(
  117                         Token{.type = TokenType::Identifier, .lexeme = word, .location = location});
  118                     previousTokenIsGoto = false;
  119                 }
  120 
  121                 break;
  122             }
  123 
  124             case State::Number:
  125             {
  126                 std::string number;
  127 
  128                 while (i < source_.size() && std::isdigit(static_cast<unsigned char>(source_[i])))
  129                 {
  130                     number += advance();
  131                 }
  132 
  133                 if (i < source_.size() &&
  134                     (std::isalpha(static_cast<unsigned char>(source_[i])) || source_[i] == '_'))
  135                 {
  136                     throw std::runtime_error("scanner error at line " + std::to_string(line) +
  137                                              ", column " + std::to_string(column) +
  138                                              ": number cannot be followed by identifier");
  139                 }
  140 
  141                 bool isLabel = previousTokenIsGoto || (i < source_.size() && source_[i] == ':');
  142 
  143                 tokens.push_back(Token{
  144                     .type = isLabel ? TokenType::Label : TokenType::Constant,
  145                     .lexeme = number,
  146                     .location = location,
  147                 });
  148 
  149                 previousTokenIsGoto = false;
  150                 break;
  151             }
  152 
  153             case State::OperatorOrPunctuation:
  154             {
  155                 if (ch == '>')
  156                 {
  157                     advance();
  158 
  159                     if (i < source_.size() && source_[i] == '=')
  160                     {
  161                         advance();
  162 
  163                         tokens.push_back(Token{
  164                             .type = TokenType::GreaterEqual,
  165                             .lexeme = ">=",
  166                             .location = location,
  167                         });
  168                     }
  169                     else
  170                     {
  171                         tokens.push_back(
  172                             Token{.type = TokenType::Greater, .lexeme = ">", .location = location});
  173                     }
  174 
  175                     previousTokenIsGoto = false;
  176                     break;
  177                 }
  178 
  179                 if (ch == '<')
  180                 {
  181                     advance();
  182 
  183                     if (i < source_.size() && source_[i] == '=')
  184                     {
  185                         advance();
  186 
  187                         tokens.push_back(Token{
  188                             .type = TokenType::LessEqual,
  189                             .lexeme = "<=",
  190                             .location = location,
  191                         });
  192                     }
  193                     else if (i < source_.size() && source_[i] == '>')
  194                     {
  195                         advance();
  196 
  197                         tokens.push_back(Token{
  198                             .type = TokenType::NotEqual,
  199                             .lexeme = "<>",
  200                             .location = location,
  201                         });
  202                     }
  203                     else
  204                     {
  205                         tokens.push_back(
  206                             Token{.type = TokenType::Less, .lexeme = "<", .location = location});
  207                     }
  208 
  209                     previousTokenIsGoto = false;
  210                     break;
  211                 }
  212 
  213                 switch (ch)
  214                 {
  215                     case '+':
  216                         advance();
  217                         tokens.push_back(
  218                             Token{.type = TokenType::Plus, .lexeme = "+", .location = location});
  219                         break;
  220 
  221                     case '-':
  222                         advance();
  223                         tokens.push_back(
  224                             Token{.type = TokenType::Minus, .lexeme = "-", .location = location});
  225                         break;
  226 
  227                     case '*':
  228                         advance();
  229                         tokens.push_back(
  230                             Token{.type = TokenType::Star, .lexeme = "*", .location = location});
  231                         break;
  232 
  233                     case '/':
  234                         advance();
  235                         tokens.push_back(
  236                             Token{.type = TokenType::Slash, .lexeme = "/", .location = location});
  237                         break;
  238 
  239                     case '^':
  240                         advance();
  241                         tokens.push_back(
  242                             Token{.type = TokenType::Power, .lexeme = "^", .location = location});
  243                         break;
  244 
  245                     case '=':
  246                         advance();
  247                         tokens.push_back(
  248                             Token{.type = TokenType::Assign, .lexeme = "=", .location = location});
  249                         break;
  250 
  251                     case '(':
  252                         advance();
  253                         tokens.push_back(Token{
  254                             .type = TokenType::LeftParen,
  255                             .lexeme = "(",
  256                             .location = location,
  257                         });
  258                         break;
  259 
  260                     case ')':
  261                         advance();
  262                         tokens.push_back(Token{
  263                             .type = TokenType::RightParen,
  264                             .lexeme = ")",
  265                             .location = location,
  266                         });
  267                         break;
  268 
  269                     case ':':
  270                         advance();
  271                         tokens.push_back(
  272                             Token{.type = TokenType::Colon, .lexeme = ":", .location = location});
  273                         break;
  274 
  275                     case ';':
  276                         advance();
  277                         tokens.push_back(Token{
  278                             .type = TokenType::Semicolon,
  279                             .lexeme = ";",
  280                             .location = location,
  281                         });
  282                         break;
  283 
  284                     default:
  285                         throw std::runtime_error(
  286                             "scanner error at line " + std::to_string(location.line) + ", column " +
  287                             std::to_string(location.column) + ": unknown character: " + ch);
  288                 }
  289 
  290                 previousTokenIsGoto = false;
  291                 break;
  292             }
  293         }
  294     }
  295 
  296     if (tokens.empty() || tokens.back().type != TokenType::End)
  297     {
  298         tokens.push_back(Token{
  299             .type = TokenType::End,
  300             .lexeme = "#",
  301             .location = SourceLocation{.line = line, .column = column},
  302         });
  303     }
  304 
  305     return tokens;
  306 }
