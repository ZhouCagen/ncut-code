   11 Parser::Parser(std::vector<Token> tokens) : tokens_(std::move(tokens))
   12 {
   13     if (tokens_.empty() || tokens_.back().type != TokenType::End)
   14     {
   15         tokens_.push_back(Token{
   16             .type = TokenType::End,
   17             .lexeme = "#",
   18             .location = SourceLocation{},
   19         });
   20     }
   21 }

   23 void Parser::parseProgram()
   24 {
   25     current_ = 0;
   26     parseSuccess_ = false;
   27     reductions_ = ReductionRecorder{};
   28 
   29     while (tokens_[current_].type == TokenType::Semicolon)
   30         current_++;
   31 
   32     while (tokens_[current_].type != TokenType::End)
   33     {
   34         parseStatement();
   35 
   36         while (tokens_[current_].type == TokenType::Semicolon)
   37             current_++;
   38     }
   39 
   40     reductions_.record(Production::Program);
   41     parseSuccess_ = true;
   42 }

   44 void Parser::parseStatement()
   45 {
   46     switch (tokens_[current_].type)
   47     {
   48         case TokenType::Identifier:
   49             parseAssignment();
   50             reductions_.record(Production::StatementToAssignment);
   51             return;
   52 
   53         case TokenType::Label:
   54             parseLabeledAssignment();
   55             reductions_.record(Production::StatementToLabeledAssignment);
   56             return;
   57 
   58         case TokenType::If:
   59             parseIfStatement();
   60             reductions_.record(Production::StatementToIf);
   61             return;
   62 
   63         case TokenType::Goto:
   64             parseGotoStatement();
   65             reductions_.record(Production::StatementToGoto);
   66             return;
   67 
   68         default:
   69         {
   70             const Token &token = tokens_[current_];
   71 
   72             throw ParseError(token.location, "expected statement, got `" + token.lexeme + "`");
   73         }
   74     }
   75 }

   77 void Parser::parseAssignment()
   78 {
   79     if (tokens_[current_].type != TokenType::Identifier)
   80     {
   81         const Token &token = tokens_[current_];
   82 
   83         throw ParseError(token.location,
   84                          "expected identifier at beginning of assignment statement, got `" +
   85                              token.lexeme + "`");
   86     }
   87 
   88     current_++;
   89 
   90     if (tokens_[current_].type != TokenType::Assign)
   91     {
   92         const Token &token = tokens_[current_];
   93 
   94         throw ParseError(token.location, "expected = after identifier, got `" + token.lexeme + "`");
   95     }
   96 
   97     current_++;
   98 
   99     parseArithmeticExpression();
  100 
  101     reductions_.record(Production::Assignment);
  102 }

  104 void Parser::parseLabeledAssignment()
  105 {
  106     if (tokens_[current_].type != TokenType::Label)
  107     {
  108         const Token &token = tokens_[current_];
  109 
  110         throw ParseError(token.location, "expected label, got `" + token.lexeme + "`");
  111     }
  112 
  113     current_++;
  114 
  115     if (tokens_[current_].type != TokenType::Colon)
  116     {
  117         const Token &token = tokens_[current_];
  118 
  119         throw ParseError(token.location, "expected : after label, got `" + token.lexeme + "`");
  120     }
  121 
  122     current_++;
  123 
  124     parseAssignment();
  125 
  126     reductions_.record(Production::LabeledAssignment);
  127 }

  129 void Parser::parseIfStatement()
  130 {
  131     if (tokens_[current_].type != TokenType::If)
  132     {
  133         const Token &token = tokens_[current_];
  134 
  135         throw ParseError(token.location, "expected IF, got `" + token.lexeme + "`");
  136     }
  137 
  138     current_++;
  139 
  140     bool hasOuterParentheses = false;
  141 
  142     if (tokens_[current_].type == TokenType::LeftParen)
  143     {
  144         hasOuterParentheses = true;
  145         current_++;
  146     }
  147 
  148     parseBoolExpression();
  149 
  150     if (hasOuterParentheses)
  151     {
  152         if (tokens_[current_].type != TokenType::RightParen)
  153         {
  154             const Token &token = tokens_[current_];
  155 
  156             throw ParseError(token.location,
  157                              "expected ) after boolean expression, got `" + token.lexeme + "`");
  158         }
  159 
  160         current_++;
  161     }
  162 
  163     if (tokens_[current_].type != TokenType::Then)
  164     {
  165         const Token &token = tokens_[current_];
  166 
  167         throw ParseError(token.location,
  168                          "expected THEN after boolean expression, got `" + token.lexeme + "`");
  169     }
  170 
  171     current_++;
  172 
  173     parseStatement();
  174 
  175     while (tokens_[current_].type == TokenType::Semicolon)
  176         current_++;
  177 
  178     if (tokens_[current_].type == TokenType::Else)
  179     {
  180         current_++;
  181 
  182         parseStatement();
  183 
  184         reductions_.record(Production::IfThenElse);
  185     }
  186     else
  187     {
  188         reductions_.record(Production::IfThen);
  189     }
  190 }

  192 void Parser::parseGotoStatement()
  193 {
  194     if (tokens_[current_].type != TokenType::Goto)
  195     {
  196         const Token &token = tokens_[current_];
  197 
  198         throw ParseError(token.location, "expected GOTO, got `" + token.lexeme + "`");
  199     }
  200 
  201     current_++;
  202 
  203     if (tokens_[current_].type != TokenType::Label)
  204     {
  205         const Token &token = tokens_[current_];
  206 
  207         throw ParseError(token.location, "expected label after GOTO, got `" + token.lexeme + "`");
  208     }
  209 
  210     current_++;
  211 
  212     reductions_.record(Production::Goto);
  213 }

  215 void Parser::parseBoolExpression()
  216 {
  217     parseArithmeticExpression();
  218 
  219     if (!isRelationalOperator(tokens_[current_].type))
  220     {
  221         const Token &token = tokens_[current_];
  222 
  223         throw ParseError(token.location,
  224                          "expected relational operator in boolean expression, got `" +
  225                              token.lexeme + "`");
  226     }
  227 
  228     current_++;
  229 
  230     parseArithmeticExpression();
  231 
  232     reductions_.record(Production::BoolExpression);
  233 }

  235 void Parser::parseArithmeticExpression(int minPrecedence)
  236 {
  237     parsePrimary();
  238 
  239     while (tokens_[current_].type != TokenType::End)
  240     {
  241         TokenType operatorType = tokens_[current_].type;
  242 
  243         if (operatorType == TokenType::Identifier || operatorType == TokenType::Label ||
  244             operatorType == TokenType::If || operatorType == TokenType::Goto ||
  245             operatorType == TokenType::Then || operatorType == TokenType::Else ||
  246             operatorType == TokenType::RightParen || operatorType == TokenType::Colon ||
  247             operatorType == TokenType::Semicolon || isRelationalOperator(operatorType))
  248         {
  249             break;
  250         }
  251 
  252         int operatorPrecedence = -1;
  253 
  254         switch (operatorType)
  255         {
  256             case TokenType::Plus:
  257             case TokenType::Minus:
  258                 operatorPrecedence = 1;
  259                 break;
  260 
  261             case TokenType::Star:
  262             case TokenType::Slash:
  263                 operatorPrecedence = 2;
  264                 break;
  265 
  266             case TokenType::Power:
  267                 operatorPrecedence = 3;
  268                 break;
  269 
  270             default:
  271             {
  272                 const Token &token = tokens_[current_];
  273 
  274                 throw ParseError(token.location,
  275                                  "invalid arithmetic operator, got `" + token.lexeme + "`");
  276             }
  277         }
  278 
  279         if (operatorPrecedence < minPrecedence)
  280             break;
  281 
  282         Token operatorToken = tokens_[current_];
  283         current_++;
  284 
  285         int nextMinPrecedence = operatorPrecedence + 1;
  286 
  287         if (operatorType == TokenType::Power)
  288             nextMinPrecedence = operatorPrecedence;
  289 
  290         parseArithmeticExpression(nextMinPrecedence);
  291 
  292         switch (operatorType)
  293         {
  294             case TokenType::Plus:
  295                 reductions_.record(Production::Add);
  296                 break;
  297 
  298             case TokenType::Minus:
  299                 reductions_.record(Production::Subtract);
  300                 break;
  301 
  302             case TokenType::Star:
  303                 reductions_.record(Production::Multiply);
  304                 break;
  305 
  306             case TokenType::Slash:
  307                 reductions_.record(Production::Divide);
  308                 break;
  309 
  310             case TokenType::Power:
  311                 reductions_.record(Production::Power);
  312                 break;
  313 
  314             default:
  315                 throw ParseError(operatorToken.location,
  316                                  "invalid arithmetic operator, got `" + operatorToken.lexeme + "`");
  317         }
  318     }
  319 }

  321 void Parser::parsePrimary()
  322 {
  323     switch (tokens_[current_].type)
  324     {
  325         case TokenType::Identifier:
  326             current_++;
  327             reductions_.record(Production::FactorIdentifier);
  328             return;
  329 
  330         case TokenType::Constant:
  331             current_++;
  332             reductions_.record(Production::FactorConstant);
  333             return;
  334 
  335         case TokenType::LeftParen:
  336         {
  337             current_++;
  338 
  339             parseArithmeticExpression();
  340 
  341             if (tokens_[current_].type != TokenType::RightParen)
  342             {
  343                 const Token &token = tokens_[current_];
  344 
  345                 throw ParseError(token.location, "expected ) after arithmetic expression, got `" +
  346                                                      token.lexeme + "`");
  347             }
  348 
  349             current_++;
  350 
  351             reductions_.record(Production::FactorParenthesized);
  352             return;
  353         }
  354 
  355         default:
  356         {
  357             const Token &token = tokens_[current_];
  358 
  359             throw ParseError(token.location,
  360                              "expected identifier, constant, or parenthesized expression, got `" +
  361                                  token.lexeme + "`");
  362         }
  363     }
  364 }

  371 void Parser::writeResult(const std::string &path) const
  372 {
  373     fs::path outputPath(path);
  374 
  375     if (!outputPath.parent_path().empty())
  376         fs::create_directories(outputPath.parent_path());
  377 
  378     std::ofstream output(path);
  379 
  380     if (!output.is_open())
  381         throw std::runtime_error("cannot create parser result file: " + path);
  382 
  383     if (parseSuccess_)
  384     {
  385         output << "Syntax analysis succeeded.\n";
  386         output << "Reduction count: " << reductions_.reductions().size() << '\n';
  387     }
  388     else
  389     {
  390         output << "Syntax analysis failed.\n";
  391     }
