    3 #include <cstddef>

    6 
    7 enum class TokenType
    8 {
    9     Identifier, // 标识符
   10     Constant,   // 常数
   11     Label,      // 标号
   12 
   13     If,   // IF
   14     Then, // THEN
   15     Else, // ELSE
   16     Goto, // GOTO
   17 
   18     Plus,  // +
   19     Minus, // -
   20     Star,  // *
   21     Slash, // /
   22     Power, // ^ 或 ↑
   23 
   24     Assign, // =
   25 
   26     Greater,      // >
   27     Less,         // <
   28     GreaterEqual, // >=
   29     LessEqual,    // <=
   30     NotEqual,     // <>
   31 
   32     LeftParen,  // (
   33     RightParen, // )
   34     Colon,      // :
   35     Semicolon,  // ;
   36     End         // #
   37 };
   38 
   39 struct SourceLocation
   40 {
   41     std::size_t line = 1;
   42     std::size_t column = 1;
   43 };
   44 
   45 struct Token
   46 {
   47     TokenType type;
   48     std::string lexeme;
   49     SourceLocation location;
   50 };
   51 
   52 inline std::string_view tokenTypeName(TokenType type)
   53 {
   54     switch (type)
   55     {
   56         case TokenType::Identifier:
   57             return "Identifier";
   58         case TokenType::Constant:
   59             return "Constant";
   60         case TokenType::Label:
   61             return "Label";
   62         case TokenType::If:
   63             return "IF";
   64         case TokenType::Then:
   65             return "THEN";
   66         case TokenType::Else:
   67             return "ELSE";
   68         case TokenType::Goto:
   69             return "GOTO";
   70         case TokenType::Plus:
   71             return "+";
   72         case TokenType::Minus:
   73             return "-";
   74         case TokenType::Star:
   75             return "*";
   76         case TokenType::Slash:
   77             return "/";
   78         case TokenType::Power:
   79             return "^";
   80         case TokenType::Assign:
   81             return "=";
   82         case TokenType::Greater:
   83             return ">";
   84         case TokenType::Less:
   85             return "<";
   86         case TokenType::GreaterEqual:
   87             return ">=";
   88         case TokenType::LessEqual:
   89             return "<=";
   90         case TokenType::NotEqual:
   91             return "<>";
   92         case TokenType::LeftParen:
   93             return "(";
   94         case TokenType::RightParen:
   95             return ")";
   96         case TokenType::Colon:
   97             return ":";
   98         case TokenType::Semicolon:
   99             return ";";
  100         case TokenType::End:
  101             return "#";
  102     }
  103 
  104     return "Unknown";
  105 }
  106 
  107 inline bool isRelationalOperator(TokenType type)
  108 {
  109     return type == TokenType::Greater || type == TokenType::Less ||
  110            type == TokenType::GreaterEqual || type == TokenType::LessEqual ||
  111            type == TokenType::Assign || type == TokenType::NotEqual;
  112 }
