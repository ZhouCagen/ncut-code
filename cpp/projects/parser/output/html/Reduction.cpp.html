    7 namespace fs = std::filesystem;

    9 int productionNumber(Production production)
   10 {
   11     return static_cast<int>(production);
   12 }
   13 
   14 std::string productionDescription(Production production)
   15 {
   16     switch (production)
   17     {
   18         case Production::Program:
   19             return "Program -> StatementList #";
   20 
   21         case Production::StatementToAssignment:
   22             return "Statement -> Assignment";
   23         case Production::StatementToIf:
   24             return "Statement -> IfStatement";
   25         case Production::StatementToGoto:
   26             return "Statement -> GotoStatement";
   27         case Production::StatementToLabeledAssignment:
   28             return "Statement -> LabeledAssignment";
   29 
   30         case Production::LabeledAssignment:
   31             return "LabeledAssignment -> Label : Assignment";
   32         case Production::Assignment:
   33             return "Assignment -> Identifier = ArithmeticExpression";
   34 
   35         case Production::IfThen:
   36             return "IfStatement -> IF BoolExpression THEN Statement";
   37         case Production::IfThenElse:
   38             return "IfStatement -> IF BoolExpression THEN Statement ELSE Statement";
   39 
   40         case Production::Goto:
   41             return "GotoStatement -> GOTO Label";
   42 
   43         case Production::BoolExpression:
   44             return "BoolExpression -> ArithmeticExpression Relop ArithmeticExpression";
   45 
   46         case Production::Add:
   47             return "ArithmeticExpression -> ArithmeticExpression + Term";
   48         case Production::Subtract:
   49             return "ArithmeticExpression -> ArithmeticExpression - Term";
   50         case Production::Multiply:
   51             return "Term -> Term * Factor";
   52         case Production::Divide:
   53             return "Term -> Term / Factor";
   54         case Production::Power:
   55             return "Term -> Factor ^ Term";
   56 
   57         case Production::FactorIdentifier:
   58             return "Factor -> Identifier";
   59         case Production::FactorConstant:
   60             return "Factor -> Constant";
   61         case Production::FactorParenthesized:
   62             return "Factor -> ( ArithmeticExpression )";
   63     }
   64 
   65     return "Unknown production";
   66 }
   67 
   68 void ReductionRecorder::record(Production production)
   69 {
   70     reductions_.push_back(
   71         Reduction{.production = production, .description = productionDescription(production)});
   72 }
   73 
   74 void ReductionRecorder::writeToFile(const std::string &path) const
   75 {
   76     fs::path outputPath(path);
   77 
   78     if (!outputPath.parent_path().empty())
   79         fs::create_directories(outputPath.parent_path());
   80 
   81     std::ofstream output(path);
   82 
   83     if (!output.is_open())
   84         throw std::runtime_error("cannot create reduction file: " + path);
   85 
   86     output << "Reduction sequence:\n";
   87 
   88     for (const auto &reduction : reductions_)
   89     {
   90         output << productionNumber(reduction.production) << ". " << reduction.description << '\n';
   91     }
   92 }
   93 
   94 const std::vector<Reduction> &ReductionRecorder::reductions() const
   95 {
   96     return reductions_;
   97 }
